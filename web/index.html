Tài liệu REST API

# 1. Kiểm tra trạng thái Cookie & Auto-Rotate
GET /api/session/status

# 2. Tạo 1 cảnh video 15s kèm ảnh Thành phần (StructuredImage)
POST /api/generate-scene
{
  "prompt": "Continuous unbroken 15-second cinematic shot...",
  "duration": 15,
  "image": "assets_1080p/asset_01_kenh_ket_noi_1080p.png",
  "output_filename": "scene1_15s_1080p.mp4",
  "upscale_1080p": true
}

# 3. Sản xuất trọn gói TVC 30s/60s (15s/cảnh + 1-Take TTS + Lyria BGM)
POST /api/produce-commercial
{
  "output_filename": "tvc_master_1080p.mp4",
  "scene_duration": 15,
  "voice": "Puck",
  "bpm": 116,
  "scale": "MINOR",
  "scenes": [ ... ]
}
